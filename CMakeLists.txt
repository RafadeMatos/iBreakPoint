cmake_minimum_required(VERSION 3.16)
project(ibreakpoint VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)
option(IBP_BUILD_TESTS "Build tests" ON)
add_executable(ibreakpoint
    src/main.c src/cli.c src/ui.c src/device.c src/doctor.c src/analyze.c src/serial.c src/protocol.c)
target_include_directories(ibreakpoint PRIVATE include)
target_compile_options(ibreakpoint PRIVATE -Wall -Wextra -Wpedantic -Wformat=2 -Wshadow)
target_compile_definitions(ibreakpoint PRIVATE IBP_ASSET_DIR="${CMAKE_INSTALL_PREFIX}/share/ibreakpoint")
install(TARGETS ibreakpoint RUNTIME DESTINATION bin)
install(DIRECTORY assets/ DESTINATION share/ibreakpoint/assets)
install(FILES docs/protocol.md DESTINATION share/doc/ibreakpoint)
if(IBP_BUILD_TESTS)
  enable_testing()
  add_executable(ibp_tests tests/test_core.c src/protocol.c src/ui.c)
  target_include_directories(ibp_tests PRIVATE include)
  target_compile_options(ibp_tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  add_test(NAME ibp_core COMMAND ibp_tests)
endif()
